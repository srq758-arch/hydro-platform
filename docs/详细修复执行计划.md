# 文件: hydro_platform/app/queries.py

def list_review_items(self, status: str = None) -> list[dict]:
    """列出待复核项（D02修复：从candidates读取）"""
    query = """
        SELECT 
            ri.review_id,
            ri.entity_id,
            ri.fact_type,
            ri.fact_key,
            ri.candidate_id,  -- ✓ 必须包含
            ri.status,
            ri.reason,
            ri.created_at,
            s.canonical_name,
            s.country,
            ec.generation_gwh,
            ec.period_label,
            ec.document_id,
            d.title as source_title
        FROM review_items ri
        JOIN stations s ON ri.entity_id = s.entity_id
        LEFT JOIN extraction_candidates ec ON ri.candidate_id = ec.candidate_id  -- ✓ 关键JOIN
        LEFT JOIN documents d ON ec.document_id = d.document_id
        WHERE 1=1
    """
    
    if status:
        query += " AND ri.status = ?"
        cursor = self.conn.execute(query, (status,))
    else:
        cursor = self.conn.execute(query)
    
    items = [dict(row) for row in cursor.fetchall()]
    return items
```

# 文件: hydro_platform/pipeline/orchestrator.py

def apply_review_decision(
    conn: sqlite3.Connection,
    review_id: str,
    action: str,  # approve / reject / request_more_evidence
    reviewer: str = "system"
) -> dict:
    """应用复核决定（D03修复：补证不触发发布）"""
    
    if action not in ['approve', 'reject', 'request_more_evidence']:
        raise ValueError(f"非法action: {action}")
    
    # 读取review_item
    cursor = conn.execute(
        "SELECT * FROM review_items WHERE review_id = ?",
        (review_id,)
    )
    item = cursor.fetchone()
    if not item:
        raise ValueError(f"review_id不存在: {review_id}")
    
    if action == 'request_more_evidence':
        # ✓ 补证：只更新状态，不创建generation_records
        conn.execute("""
            UPDATE review_items
            SET status = 'awaiting_evidence',
                reviewer = ?,
                resolved_at = datetime('now')
            WHERE review_id = ?
        """, (reviewer, review_id))
        
        # 记录事件
        conn.execute("""
            INSERT INTO review_events (
                event_id, review_id, action, reviewer, created_at
            ) VALUES (?, ?, ?, ?, datetime('now'))
        """, (f"evt_{review_id}", review_id, action, reviewer))
        
        conn.commit()
        return {"status": "awaiting_evidence", "published": False}
    
    elif action == 'approve':
        # ✓ 批准：创建generation_records
        # ... 批准逻辑
        return {"status": "approved", "published": True}
    
    elif action == 'reject':
        # ✓ 拒绝：标记rejected
        # ... 拒绝逻辑
        return {"status": "rejected", "published": False}
```
